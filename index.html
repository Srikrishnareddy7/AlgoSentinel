# AlgoSentinel — Algorand Smart Contract (PyTEAL AVM v8) # Stores monitoring events, alerts, and risk scores on-chain from pyteal import * # ── Global State (4 keys) ──────────────────────────── # alert_count : uint64 — total alerts logged # last_block : uint64 — last Algorand block scanned # agent_active : uint64 — 1=ON, 0=OFF # total_txns_scanned: uint64 — transactions analyzed # ── Local State per Wallet (3 keys) ───────────────── # user_alerts : uint64 — alerts for this wallet # risk_score : uint64 — 0-100 current risk # last_alert_round : uint64 — block of last alert def approval_program(): on_creation = Seq([ App.globalPut(Bytes("alert_count"), Int(0)), App.globalPut(Bytes("last_block"), Global.round()), App.globalPut(Bytes("agent_active"), Int(1)), App.globalPut(Bytes("total_txns"), Int(0)), App.globalPut(Bytes("version"), Bytes("4.0.0")), Approve(), ]) on_log_alert = Seq([ Assert(Txn.application_args.length() == Int(3)), Assert(App.globalGet(Bytes("agent_active")) == Int(1)), App.globalPut(Bytes("alert_count"), App.globalGet(Bytes("alert_count")) + Int(1)), App.globalPut(Bytes("last_block"), Global.round()), # ARC-28 event emission for Indexer pickup Log(Concat( Bytes("ALERT:"), Txn.application_args[0], Bytes(":SEV:"), Txn.application_args[1], Bytes(":"), Txn.application_args[2], )), Approve(), ]) on_optin = Seq([ App.localPut(Txn.sender(), Bytes("user_alerts"), Int(0)), App.localPut(Txn.sender(), Bytes("risk_score"), Int(50)), App.localPut(Txn.sender(), Bytes("last_alert_round"), Global.round()), Log(Concat(Bytes("SUBSCRIBED:"), Txn.sender())), Approve(), ]) return Cond( [Txn.application_id() == Int(0), on_creation], [Txn.on_completion() == OnComplete.OptIn, on_optin], [Txn.application_args[0] == Bytes("log_alert"), on_log_alert], [Txn.application_args[0] == Bytes("update_risk"), on_update_risk], [Txn.application_args[0] == Bytes("get_status"), on_get_status], )
